// ================= VARIABEL INPUT PASIEN =================
char jenis_kelamin = 'L';  
int umur_bulan = 0;       

bool tungguInput = true; 
unsigned long lastPrintTime = 0;

void setup() {
  Serial.begin(57600);
  lcd.init();
  lcd.backlight();
  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);

  LoadCell.begin();
  LoadCell.start(2000, true);
  if (LoadCell.getTareTimeoutFlag()) {
    lcd.setCursor(0,0);
    lcd.print("HX711 Error");
    while(1);
  }
  LoadCell.setCalFactor(calibrationValue);

  // Tampilan awal
  lcd.clear();
  lcd.setCursor(0,0); lcd.print("Input L/P & Umur");
  lcd.setCursor(0,1); lcd.print("Contoh: L12");
  
  Serial.println("\n=== SISTEM PENGUKUR STUNTING & GIZI ===");
  Serial.println("STATUS: MENUNGGU PASIEN BARU");
  Serial.println("Ketik input: L12 (Laki-laki 12 bulan) atau P5 (Perempuan 5 bulan)");
  Serial.println("========================================\n");
  delay(1000);
}

void loop() {
  // ================= 1. BACA INPUT SERIAL =================
  if (Serial.available() > 0) {
    String inputData = Serial.readStringUntil('\n'); 
    inputData.trim(); 

    if (inputData.equalsIgnoreCase("S")) {
        tungguInput = true; 
        Serial.println("\n[!] PENGUKURAN SELESAI [!]");
        Serial.println("\n=== MENUNGGU PASIEN BARU ===");
        Serial.println("Silakan ketik data berikutnya (Contoh: L12 atau P5):");
        
        lcd.clear();
        lcd.setCursor(0,0); lcd.print("Input L/P & Umur");
        lcd.setCursor(0,1); lcd.print("Contoh: L12");
    }
    else if (inputData.length() >= 2) {
      char jk = inputData.charAt(0); 
      
      if (jk == 'L' || jk == 'l' || jk == 'P' || jk == 'p') {
        jenis_kelamin = jk;
        String umur_str = inputData.substring(1); 
        umur_bulan = umur_str.toInt();
        
        tungguInput = false; 
        
        Serial.println("\n[!] DATA PASIEN BERHASIL DITERIMA [!]");
        Serial.print("Jenis Kelamin : "); Serial.println(jenis_kelamin);
        Serial.print("Umur (Bulan)  : "); Serial.println(umur_bulan);
        Serial.println("-> SEDANG MENGUKUR... (Ketik 'S' jika sudah selesai)\n");
      } else {
        Serial.println("\n[X] Format Salah! Awali dengan huruf L atau P [X]\n");
      }
    }
  }

  // ================= 2. PROSES SENSOR =================
  if (LoadCell.update()) {
    if (millis() - lastPrintTime >= 500) {
      lastPrintTime = millis();

      if (!tungguInput) {
          // --- Baca Sensor Berat ---
          float weightGram = LoadCell.getData();
          float weightKg = weightGram / 1000.0;
          if (weightKg < 0) weightKg = 0; 
    
          // --- Baca Sensor Tinggi ---
          digitalWrite(TRIG_PIN, LOW); delayMicroseconds(2);
          digitalWrite(TRIG_PIN, HIGH); delayMicroseconds(10);
          digitalWrite(TRIG_PIN, LOW);
          long duration = pulseIn(ECHO_PIN, HIGH, 30000);
          float distance = duration * 0.034 / 2.0;
          float height = MAX_HEIGHT - distance;
          if (height < 0) height = 0;
          if (height > MAX_HEIGHT) height = MAX_HEIGHT;
    
          // --- Ambil Data Base ---
          int umur_aman = umur_bulan;
          if (umur_aman > 24) umur_aman = 24; 
          if (umur_aman < 0) umur_aman = 0;
    
          float M_tinggi = 0, S_tinggi = 0;
          float L_berat = 0, M_berat = 0, S_berat = 0;
    
          if (jenis_kelamin == 'L' || jenis_kelamin == 'l') {
              M_tinggi = panjangLaki[umur_aman].M; S_tinggi = panjangLaki[umur_aman].S;
              L_berat = beratLaki[umur_aman].L; M_berat = beratLaki[umur_aman].M; S_berat = beratLaki[umur_aman].S;
          } else {
              M_tinggi = panjangPerempuan[umur_aman].M; S_tinggi = panjangPerempuan[umur_aman].S;
              L_berat = beratPerempuan[umur_aman].L; M_berat = beratPerempuan[umur_aman].M; S_berat = beratPerempuan[umur_aman].S;
          }
    
          // --- Kalkulasi Z-Score ---
          float z_tinggi = ((height / M_tinggi) - 1.0) / S_tinggi;
          float z_berat = (pow((weightKg / M_berat), L_berat) - 1.0) / (S_berat * L_berat);
    
          // --- Tentukan Status ---
          String status_tinggi = "";
          if (z_tinggi < -3.0) status_tinggi = "SgtPndk"; 
          else if (z_tinggi < -2.0) status_tinggi = "Stunting";
          else status_tinggi = "Normal";

          String status_berat = "";
          if (z_berat < -3.0) status_berat = "GzBuruk"; 
          else if (z_berat < -2.0) status_berat = "GzKurang";
          else status_berat = "Normal";
    
          // --- Tampil ke Serial ---
          Serial.print("Pasien: "); Serial.print(jenis_kelamin); Serial.print(umur_aman);
          Serial.print(" | Tinggi: "); Serial.print(height, 1);
          Serial.print("cm (Z: "); Serial.print(z_tinggi, 2); Serial.print(" "); Serial.print(status_tinggi);
          Serial.print(") | Berat: "); Serial.print(weightKg, 2);
          Serial.print("kg (Z: "); Serial.print(z_berat, 2); Serial.print(" "); Serial.println(status_berat);
    
          // --- Tampil ke LCD ---
          // Format LCD Baris 1: T:-2.1 Stunting
          // Format LCD Baris 2: B:-1.5 GzKurang
          lcd.clear();
          lcd.setCursor(0,0);
          lcd.print("T:"); lcd.print(z_tinggi, 1); lcd.print(" "); lcd.print(status_tinggi);
          lcd.setCursor(0,1);
          lcd.print("B:"); lcd.print(z_berat, 1); lcd.print(" "); lcd.print(status_berat);
      }
    }
  }
}
