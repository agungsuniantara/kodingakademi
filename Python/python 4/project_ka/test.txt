#include <HX711_ADC.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <EEPROM.h>

// ================= HX711 =================
const int HX711_dout = 4;
const int HX711_sck = 5;
HX711_ADC LoadCell(HX711_dout, HX711_sck);

// ================= Ultrasonic =================
#define TRIG_PIN 8
#define ECHO_PIN 9
#define MAX_HEIGHT 100.0 

// ================= LCD =================
LiquidCrystal_I2C lcd(0x27, 16, 2);

float calibrationValue = 22.55;

// ================= DATABASE WHO LENGKAP (0 - 24 Bulan) =================
struct DataWHO {
  float M;
  float S;
};

// Data Laki-laki (0 - 24 Bulan)
DataWHO dataLaki[25] = {

// Data Perempuan (0 - 24 Bulan)
DataWHO dataPerempuan[25] = {
  {49.1477, 0.03790}, {53.6872, 0.03640}, {57.0673, 0.03568}, {59.8029, 0.03520}, {62.0899, 0.03486},

  {82.7036, 0.03643}, {83.6654, 0.03666}, {84.6040, 0.03688}, {85.5202, 0.03711}, {86.4153, 0.03734}
};

// ================= VARIABEL INPUT PASIEN (Nilai Awal) =================
char jenis_kelamin = 'L';  
int umur_bulan = 0;       

// PENANDA STATUS: True berarti sistem menahan proses dan menunggu input dari user
bool tungguInput = true; 

// Timer untuk membatasi kecepatan print Serial agar tidak banjir data
unsigned long lastPrintTime = 0;

void setup() {
  Serial.begin(57600);

  lcd.init();
  lcd.backlight();

  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);

  LoadCell.begin();
  LoadCell.start(2000, true);

  if (LoadCell.getTareTimeoutFlag()) {
    lcd.setCursor(0,0);
    lcd.print("HX711 Error");
    while(1);
  }

  LoadCell.setCalFactor(calibrationValue);

  // Tampilan awal menunggu input
  lcd.clear();
  lcd.setCursor(0,0);
  lcd.print("Input L/P & Umur");
  lcd.setCursor(0,1);
  lcd.print("Contoh: L12");
  
  // Panduan untuk user di Serial Monitor
  Serial.println("\n=== SISTEM PENGUKUR STUNTING ===");
  Serial.println("STATUS: MENUNGGU PASIEN BARU");
  Serial.println("Ketik input di Serial Monitor dengan format huruf dan angka tanpa spasi.");
  Serial.println("Contoh: L12 (Laki-laki 12 bulan) atau P5 (Perempuan 5 bulan)");
  Serial.println("================================\n");
  
  delay(1000);
}

void loop() {
  // ================= 1. BACA INPUT DARI SERIAL MONITOR =================
  if (Serial.available() > 0) {
    String inputData = Serial.readStringUntil('\n'); // Baca teks sampai tombol Enter ditekan
    inputData.trim(); // Hapus karakter spasi atau enter tak kasat mata di ujung teks

    // Jika user mengetik 'S' untuk menyelesaikan proses
    if (inputData.equalsIgnoreCase("S")) {
        tungguInput = true; // Kunci kembali proses
        
        Serial.println("\n[!] PENGUKURAN SELESAI [!]");
        Serial.println("\n=== MENUNGGU PASIEN BARU ===");
        Serial.println("Silakan ketik input data berikutnya (Contoh: L12 atau P5):");
        
        // Tampilkan permintaan input di LCD lagi
        lcd.clear();
        lcd.setCursor(0,0);
        lcd.print("Input L/P & Umur");
        lcd.setCursor(0,1);
        lcd.print("Contoh: L12");
    }
    // Jika user memasukkan data pasien baru (minimal 2 karakter)
    else if (inputData.length() >= 2) {
      char jk = inputData.charAt(0); // Ambil huruf pertama (Indeks 0)
      
      // Cek apakah huruf pertama valid (L atau P)
      if (jk == 'L' || jk == 'l' || jk == 'P' || jk == 'p') {
        jenis_kelamin = jk;
        
        // Ambil sisa teks mulai dari huruf kedua (Indeks 1) dan ubah jadi angka
        String umur_str = inputData.substring(1); 
        umur_bulan = umur_str.toInt();
        
        tungguInput = false; // Buka kunci proses!
        
        Serial.println("\n[!] DATA PASIEN BERHASIL DITERIMA [!]");
        Serial.print("Jenis Kelamin : "); Serial.println(jenis_kelamin);
        Serial.print("Umur (Bulan)  : "); Serial.println(umur_bulan);
        Serial.println("-> SEDANG MENGUKUR... (Ketik 'S' jika sudah selesai)\n");
        
      } else {
        Serial.println("\n[X] Format Salah! Awali dengan huruf L atau P (Contoh: L12) [X]\n");
      }
    }
  }

  // ================= 2. PROSES SENSOR DAN LOGIKA =================
  if (LoadCell.update()) {
    
    // Saya mengganti delay(500) dengan millis() agar sensor tetap responsif
    if (millis() - lastPrintTime >= 500) {
      lastPrintTime = millis();

      // HANYA JALANKAN PENGUKURAN JIKA STATUS tungguInput ADALAH FALSE
      if (!tungguInput) {
          float weightGram = LoadCell.getData();
          float weightKg = weightGram / 1000.0;
          if (weightKg < 0) weightKg = 0; // Hindari nilai minus pada timbangan kosong
    
          // ===== Ultrasonic =====
          digitalWrite(TRIG_PIN, LOW);
          delayMicroseconds(2);
          digitalWrite(TRIG_PIN, HIGH);
          delayMicroseconds(10);
          digitalWrite(TRIG_PIN, LOW);
    
          long duration = pulseIn(ECHO_PIN, HIGH, 30000);
          float distance = duration * 0.034 / 2.0;
    
          float height = MAX_HEIGHT - distance;
    
          if (height < 0) height = 0;
          if (height > MAX_HEIGHT) height = MAX_HEIGHT;
    
          // ===== KALKULASI Z-SCORE =====
          float nilai_M = 0.0;
          float nilai_S = 0.0;
    
          // Proteksi agar umur tidak melebihi index array (maksimal 24)
          int umur_aman = umur_bulan;
          if (umur_aman > 24) umur_aman = 24; 
          if (umur_aman < 0) umur_aman = 0;
    
          // Ambil data M dan S dari array
          if (jenis_kelamin == 'L' || jenis_kelamin == 'l') {
              nilai_M = dataLaki[umur_aman].M;
              nilai_S = dataLaki[umur_aman].S;
          } else {
              nilai_M = dataPerempuan[umur_aman].M;
              nilai_S = dataPerempuan[umur_aman].S;
          }
    
          // Rumus Z-Score
          float z_score = ((height / nilai_M) - 1.0) / nilai_S;
    
          // Tentukan Status Gizi
          String status = "";
          if (z_score < -3.0) {
              status = "SgtPndk"; 
          } else if (z_score < -2.0) {
              status = "Stunting";
          } else {
              status = "Normal";
          }
    
          // ===== Serial =====
          // Format print dirapikan agar mudah dibaca berulang-ulang
          Serial.print("Pasien: "); Serial.print(jenis_kelamin); Serial.print(umur_aman);
          Serial.print(" | Berat: "); Serial.print(weightKg, 2);
          Serial.print("kg | Tinggi: "); Serial.print(height, 1);
          Serial.print("cm | Z: "); Serial.print(z_score, 2);
          Serial.print(" | "); Serial.println(status);
    
          // ===== LCD =====
          lcd.clear();
          lcd.setCursor(0,0);
          lcd.print("W:");
          lcd.print(weightKg, 1); 
          lcd.print("kg H:");
          lcd.print(height, 1);
          lcd.print("cm");
    
          lcd.setCursor(0,1);
          lcd.print("Z:");
          lcd.print(z_score, 2);
          lcd.print(" ");
          lcd.print(status);
      }
    }
  }
}
